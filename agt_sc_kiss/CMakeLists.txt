cmake_minimum_required(VERSION 3.19)
project(agt_sc_kiss LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(Python3 COMPONENTS Interpreter Development REQUIRED)
find_package(pybind11 REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
get_filename_component(FIELD_SRC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SCANCONTEXT_SOURCE_DIR "${FIELD_SRC_ROOT}/external/scancontext" CACHE PATH "Locked upstream scancontext checkout")
get_filename_component(SCANCONTEXT_SOURCE_DIR "${SCANCONTEXT_SOURCE_DIR}" REALPATH)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../dependencies/scancontext.json" SC_LOCK)
string(JSON EXPECTED_COMMIT GET "${SC_LOCK}" commit)
execute_process(COMMAND git -c safe.directory=${SCANCONTEXT_SOURCE_DIR} -C ${SCANCONTEXT_SOURCE_DIR} rev-parse HEAD OUTPUT_VARIABLE ACTUAL_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT ACTUAL_COMMIT STREQUAL EXPECTED_COMMIT)
  message(FATAL_ERROR "Scan Context HEAD differs from research lock")
endif()
set(VENDOR "${CMAKE_CURRENT_BINARY_DIR}/scancontext")
file(MAKE_DIRECTORY "${VENDOR}")
foreach(NAME Scancontext.cpp Scancontext.h nanoflann.hpp KDTreeVectorOfVectorsAdaptor.h tictoc.h)
  set(INPUT "${SCANCONTEXT_SOURCE_DIR}/cpp/module/Scancontext/${NAME}")
  file(SHA256 "${INPUT}" ACTUAL_SHA)
  string(JSON EXPECTED_SHA GET "${SC_LOCK}" files "${NAME}")
  if(NOT ACTUAL_SHA STREQUAL EXPECTED_SHA)
    message(FATAL_ERROR "Scan Context source changed: ${NAME}; preserved")
  endif()
  if(NOT NAME STREQUAL "Scancontext.h")
    configure_file("${INPUT}" "${VENDOR}/${NAME}" COPYONLY)
  endif()
endforeach()
# Remove unused ROS1/OpenCV includes, without modifying the upstream checkout.
file(READ "${SCANCONTEXT_SOURCE_DIR}/cpp/module/Scancontext/Scancontext.h" HEADER)
foreach(UNUSED "#include <opencv2/opencv.hpp>" "#include <opencv2/core/eigen.hpp>" "#include <opencv2/highgui/highgui.hpp>" "#include <cv_bridge/cv_bridge.h>" "#include <pcl_conversions/pcl_conversions.h>")
  string(REPLACE "${UNUSED}" "" HEADER "${HEADER}")
endforeach()
# Fixed research profile: gravity-level mapping_body, 30 m query range.
string(REPLACE "LIDAR_HEIGHT = 2.0" "LIDAR_HEIGHT = 0.0" HEADER "${HEADER}")
string(REPLACE "PC_MAX_RADIUS = 80.0" "PC_MAX_RADIUS = 30.0" HEADER "${HEADER}")
set(OLD_HEADER "")
if(EXISTS "${VENDOR}/Scancontext.h")
  file(READ "${VENDOR}/Scancontext.h" OLD_HEADER)
endif()
if(NOT HEADER STREQUAL OLD_HEADER)
  file(WRITE "${VENDOR}/Scancontext.h" "${HEADER}")
endif()
pybind11_add_module(_scancontext src/bindings.cpp "${VENDOR}/Scancontext.cpp")
target_compile_features(_scancontext PRIVATE cxx_std_17)
target_include_directories(_scancontext PRIVATE "${VENDOR}" ${PCL_INCLUDE_DIRS})
target_link_libraries(_scancontext PRIVATE Eigen3::Eigen ${PCL_LIBRARIES})
ament_python_install_package(${PROJECT_NAME})
install(TARGETS _scancontext LIBRARY DESTINATION "${PYTHON_INSTALL_DIR}/${PROJECT_NAME}")
install(PROGRAMS scripts/sc_kiss_localizer DESTINATION lib/${PROJECT_NAME})
install(FILES NOTICE.md DESTINATION share/${PROJECT_NAME})
ament_package()
