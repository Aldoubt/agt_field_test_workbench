cmake_minimum_required(VERSION 3.8)
project(agt_field_test_bringup)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_msgs REQUIRED)
add_executable(fwmini_tf_guard src/fwmini_tf_guard.cpp)
target_compile_features(fwmini_tf_guard PUBLIC cxx_std_17)
ament_target_dependencies(fwmini_tf_guard rclcpp tf2_msgs)
install(TARGETS fwmini_tf_guard DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/localization_audit.py scripts/fwmini_tf_contract.py DESTINATION lib/${PROJECT_NAME})
ament_package()
