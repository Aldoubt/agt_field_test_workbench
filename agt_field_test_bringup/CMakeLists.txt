cmake_minimum_required(VERSION 3.8)
project(agt_field_test_bringup)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_msgs REQUIRED)
add_executable(fwmini_tf_guard src/fwmini_tf_guard.cpp)
target_compile_features(fwmini_tf_guard PUBLIC cxx_std_17)
ament_target_dependencies(fwmini_tf_guard rclcpp tf2_msgs)
install(TARGETS fwmini_tf_guard DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/localization_audit.py scripts/fwmini_tf_contract.py scripts/time_align.py ${CMAKE_CURRENT_SOURCE_DIR}/../scripts/gps_candidate_gate.py DESTINATION lib/${PROJECT_NAME})
find_package(PCL REQUIRED COMPONENTS io filters)
find_package(yaml-cpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_srvs REQUIRED)
add_executable(pcd_ops src/pcd_ops.cpp)
target_compile_features(pcd_ops PUBLIC cxx_std_17)
target_include_directories(pcd_ops PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(pcd_ops ${PCL_LIBRARIES} yaml-cpp)
add_executable(lio_artifact src/lio_artifact.cpp)
target_compile_features(lio_artifact PUBLIC cxx_std_17)
ament_target_dependencies(lio_artifact rclcpp sensor_msgs nav_msgs std_srvs)
install(TARGETS pcd_ops lio_artifact DESTINATION lib/${PROJECT_NAME})
ament_package()
