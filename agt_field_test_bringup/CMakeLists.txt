cmake_minimum_required(VERSION 3.8)
project(agt_field_test_bringup)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_msgs REQUIRED)
add_executable(fwmini_tf_guard src/fwmini_tf_guard.cpp)
target_compile_features(fwmini_tf_guard PUBLIC cxx_std_17)
ament_target_dependencies(fwmini_tf_guard rclcpp tf2_msgs)
install(TARGETS fwmini_tf_guard DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/../config/photo_navigate.xml DESTINATION share/${PROJECT_NAME}/config)
install(PROGRAMS scripts/localization_audit.py scripts/fwmini_tf_contract.py scripts/time_align.py ${CMAKE_CURRENT_SOURCE_DIR}/../scripts/gps_candidate_gate.py DESTINATION lib/${PROJECT_NAME})

find_package(std_msgs REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(agt_pointcloud_core REQUIRED)
add_executable(time_align src/time_align.cpp)
target_compile_features(time_align PUBLIC cxx_std_17)
ament_target_dependencies(time_align rclcpp sensor_msgs livox_ros_driver2)
install(TARGETS time_align DESTINATION lib/${PROJECT_NAME})
add_executable(pcd_ops src/pcd_ops.cpp)

ament_target_dependencies(livox_self_filter rclcpp livox_ros_driver2 std_msgs tf2 tf2_ros agt_pointcloud_core)
install(TARGETS pcd_ops lio_artifact livox_self_filter DESTINATION lib/${PROJECT_NAME})
find_package(nav2_smac_planner REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)

ament_target_dependencies(agt_exact_goal_planner nav2_smac_planner nav2_core nav2_costmap_2d pluginlib tf2_geometry_msgs)

  nav2_regulated_pure_pursuit_controller nav2_core nav2_costmap_2d pluginlib)
